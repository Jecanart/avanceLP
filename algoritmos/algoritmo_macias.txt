use std::collections::LinkedList;

fn main() {
    // Crear una LinkedList y añadir algunos elementos
    let mut list: LinkedList<i32> = LinkedList::new();
    list.push_back(1);
    list.push_back(2);
    list.push_back(3);
    list.push_back(4);
    list.push_back(5);

    // Iterar sobre la LinkedList usando un bucle for
    for element in list.iter() {
        println!("{}", element);
    }
}
